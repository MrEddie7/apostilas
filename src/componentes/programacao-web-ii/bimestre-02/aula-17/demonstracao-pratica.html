Exemplo Bom

<?php
declare(strict_types=1);

function dividir(float $a, float $b): float {
    if ($b === 0.0) {
        // Lança uma exceção nativa do PHP indicando argumento inválido
        throw new InvalidArgumentException("Divisão por zero não é permitida no sistema.");
    }
    return $a / $b;
}

try {
    // Código perigoso
    echo dividir(10, 2);
} catch (InvalidArgumentException $e) {
    // Código de recuperação: captura apenas erros de argumento inválido
    echo "Erro de validação: " . $e->getMessage();
} catch (Exception $e) {
    // Captura qualquer outro erro genérico que possa ter acontecido
    echo "Erro inesperado: " . $e->getMessage();
}

Carregando dados de um arquivo CSV

<?php
declare(strict_types=1);

$linhasParaExibir = [];
$mensagemErro = null;
$ponteiroArquivo = null; // Armazenará o recurso do arquivo aberto

try {
    $caminhoArquivo = __DIR__ . '/dados.csv';

    if (!file_exists($caminhoArquivo)) {
        throw new RuntimeException("O arquivo solicitado 'dados.csv' não foi encontrado no servidor.");
    }

    $ponteiroArquivo = fopen($caminhoArquivo, 'r');

    if ($ponteiroArquivo === false) {
        throw new RuntimeException("Não foi possível abrir o arquivo. Verifique as permissões de leitura.");
    }

    $numeroLinha = 0;
    while (($dados = fgetcsv($ponteiroArquivo, 1000, ";")) !== false) {
        $numeroLinha++;

        if (empty($dados) || $dados[0] === null) {
            continue;
        }

        if (count($dados) !== 4) {
            throw new ValueError("Erro de formatação na linha {$numeroLinha}: A linha não possui as 4 colunas obrigatórias.");
        }

        $linhasParaExibir[] = [
            'data'      => $dados[0],
            'descricao' => $dados[1],
            'tipo'      => $dados[2],
            'valor'     => (float) $dados[3]
        ];
    }

} catch (RuntimeException $e) {
    $mensagemErro = "Erro de Sistema: " . $e->getMessage();

} catch (ValueError $e) {
    $mensagemErro = "Erro de Dados: " . $e->getMessage();

} catch (Exception $e) {
    $mensagemErro = "Erro Inesperado: " . $e->getMessage();

} finally {
    if ($ponteiroArquivo) {
        fclose($ponteiroArquivo);
    }
}

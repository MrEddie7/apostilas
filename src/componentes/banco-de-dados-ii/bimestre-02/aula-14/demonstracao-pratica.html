create database campgames;
use campgames;

SET FOREIGN_KEY_CHECKS = 0;
DROP TABLE IF EXISTS visualizacoes_live;
DROP TABLE IF EXISTS lives;
DROP TABLE IF EXISTS streamers;
DROP TABLE IF EXISTS jogos;
SET FOREIGN_KEY_CHECKS = 1;

CREATE TABLE jogos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL,
    categoria VARCHAR(50) NOT NULL
);

CREATE TABLE streamers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nickname VARCHAR(50) NOT NULL,
    repatria_patrocinio DECIMAL(10,2) NULL,
    data_cadastro DATE NOT NULL
);

CREATE TABLE lives (
    id INT PRIMARY KEY AUTO_INCREMENT,
    id_streamer INT,
    id_jogo INT,
    duracao_minutos INT NOT NULL,
    titulo_live VARCHAR(150),
    FOREIGN KEY (id_streamer) REFERENCES streamers(id),
    FOREIGN KEY (id_jogo) REFERENCES jogos(id)
);

CREATE TABLE visualizacoes_live (
    id_live INT,
    id_usuario_espectador INT,
    minutos_assistidos INT NOT NULL,
    dispositivo VARCHAR(30),
    PRIMARY KEY (id_live, id_usuario_espectador),
    FOREIGN KEY (id_live) REFERENCES lives(id)
);

-- PRÓXIMO

SELECT 
    l.id AS numero_live,
    l.titulo_live,
    j.categoria,
    SUM(v.minutos_assistidos) AS tempo_total_audiencia
FROM lives l
INNER JOIN jogos j ON l.id_jogo = j.id
INNER JOIN visualizacoes_live v ON l.id = v.id_live
WHERE j.categoria = 'FPS' -- Filtra a origem antes de somar
GROUP BY l.id, l.titulo_live, j.categoria
HAVING SUM(v.minutos_assistidos) > 200; -- Filtra o resultado calculado
